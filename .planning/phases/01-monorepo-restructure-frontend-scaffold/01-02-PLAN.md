### Step 3 — Replace `vite.config.ts` with the proxy + alias + tailwind config

Overwrite `frontend/vite.config.ts` with EXACTLY this content (per D-09, RESEARCH.md Code Examples, PATTERNS.md):

```typescript
import path from "path"
import tailwindcss from "@tailwindcss/vite"
import react from "@vitejs/plugin-react"
import { defineConfig } from "vite"

export default defineConfig({
  plugins: [react(), tailwindcss()],
  resolve: {
    alias: {
      "@": path.resolve(__dirname, "./src"),
    },
  },
  server: {
    proxy: {
      "/api": {
        target: "http://localhost:5000",
        changeOrigin: true,
      },
      "/ask": {
        target: "http://localhost:5000",
        changeOrigin: true,
      },
      "/address-review": {
        target: "http://localhost:5000",
        changeOrigin: true,
      },
    },
  },
})
```

DO NOT add a `rewrite` function to the `/api` proxy rule — Flask expects the `/api/` prefix intact (Pitfall 3 in RESEARCH.md).

### Step 4 — Add path alias to TypeScript config

Edit `frontend/tsconfig.json` to add `baseUrl` + `paths`. The Vite template generates a file with a `references` array — keep that but inject a `compilerOptions` block:

Replace the file contents with EXACTLY:

```json
{
  "files": [],
  "references": [
    { "path": "./tsconfig.app.json" },
    { "path": "./tsconfig.node.json" }
  ],
  "compilerOptions": {
    "baseUrl": ".",
    "paths": { "@/*": ["./src/*"] }
  }
}
```

Also edit `frontend/tsconfig.app.json` — add `baseUrl` and `paths` inside its existing `compilerOptions` block so `tsc --build` honors the alias (the Vite template omits this by default). Use this merge: keep all existing compilerOptions the template generated, and ADD these two keys:

```json
    "baseUrl": ".",
    "paths": { "@/*": ["./src/*"] }
```
